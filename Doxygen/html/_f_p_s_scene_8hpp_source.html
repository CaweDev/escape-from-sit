307 void BanditTick()
308 {
309 float zoneReward[5] = {0, 0, 0, 0, 0};
310
311 for (int z = 1; z <= 4; ++z)
312 {
313 float heatSum = heatmap.SumZone(zoneMgr.GetGrid(), z);
314 float heatN = Clamp01(heatSum / heatRef);
315
316 float cctvN = Clamp01(cctvScore[z] / cctvRef);
317
318 zoneReward[z] = Clamp01(wHeat * heatN + wCCTV * cctvN);
319 }
320
321 static float switchCooldown = 0.0f;
322 switchCooldown -= banditInterval;
323 if (switchCooldown < 0.0f)
324 switchCooldown = 0.0f;
325
326 int nextArm = bandit.SelectArm();
327 if (nextArm < 0)
328 return;
329
330 int nextZone = nextArm + 1;
331
332 if (switchCooldown > 0.0f)
333 {
334 nextZone = priorityZoneId;
335 nextArm = nextZone - 1;
336 }
337 else
338 {
339 if (nextZone != priorityZoneId)
340 {
341 if (cctvScore[nextZone] > banditSwitchStrongEvidenceThreshold)
342 {
343 switchCooldown = banditSwitchCooldownStrongSec;
344 }
345 else
346 {
347 switchCooldown = banditSwitchCooldownNormalSec;
348 }
349 }
350 }
351
352 bandit.Update(nextArm, zoneReward[nextZone]);
353
354 if (nextZone == priorityZoneId)
355 {
356 priorityZoneHotTimer += banditInterval;
357 }
358 else
359 {
360 priorityZoneHotTimer = 0.0f;
361 fullEscalationActive = false;
362 }
363
364 prevChosenZoneId = nextZone;
365 priorityZoneId = nextZone;
366 ApplyPriorityZone_AssistNearest(nextZone);
367
368 for (int z = 1; z <= 4; ++z)
369 cctvScore[z] *= cctvScoreDecayFactor;
370 }
