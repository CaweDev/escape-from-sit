715 Vector2 playerPos(playerTf->position.x, playerTf->position.y);
716
717 float dx = cctvPos.x - playerPos.x;
718 float dy = cctvPos.y - playerPos.y;
719 float distance = std::sqrt(dx * dx + dy * dy);
720
721 if (distance < nearestCCTVDistance)
722 {
723 nearestCCTVDistance = distance;
724 }
725 }
726
727 float nearestTerminalDistance = cfg.distanceSentinel;
728 for (auto *terminal : cctvTerminals)
729 {
730 if (!terminal || !terminal->transform)
731 continue;
732
733 Vector2 terminalPos(terminal->transform->position.x, terminal->transform->position.y);
734 Vector2 playerPos(playerTf->position.x, playerTf->position.y);
735
736 float dx = terminalPos.x - playerPos.x;
737 float dy = terminalPos.y - playerPos.y;
738 float distance = std::sqrt(dx * dx + dy * dy);
739
740 if (distance < nearestTerminalDistance)
741 {
742 nearestTerminalDistance = distance;
743 }
744 }
745
746 // NPC proximity: separate Bully vs Annoying
747 float nearestAnnoyingDistance = cfg.distanceSentinel;
748 int annoyingCountNearby = 0;
749 float nearestBullyDistance = cfg.distanceSentinel;
750 int bullyCountNearby = 0;
751 const float npcNoticeRadius = cfg.npcNoticeRadius; // within this range we count them as 'nearby'
752 for (auto *npc : npcs)
753 {
754 if (!npc || !npc->transform)
755 continue;
756 Vector2 npcPos(npc->transform->position.x, npc->transform->position.y);
757 Vector2 playerPos(playerTf->position.x, playerTf->position.y);
758 float dx = npcPos.x - playerPos.x;
759 float dy = npcPos.y - playerPos.y;
760 float distance = std::sqrt(dx * dx + dy * dy);
761
762 if (npc->type == NPCType::Annoying)
763 {
764 if (distance < nearestAnnoyingDistance)
765 nearestAnnoyingDistance = distance;
766 if (distance <= npcNoticeRadius)
767 annoyingCountNearby++;
768 }
769 else if (npc->type == NPCType::Bully)
770 {
771 if (distance < nearestBullyDistance)
772 nearestBullyDistance = distance;
773 if (distance <= npcNoticeRadius)
774 bullyCountNearby++;
775 }
776 }
777
778 // Update guard proximity warning
779 playerHUD->UpdateGuardProximity(nearestGuardDistance);
780 // Update NPC approach indicators (Annoying + Bully)
781 playerHUD->UpdateNPCApproach(annoyingCountNearby, nearestAnnoyingDistance, bullyCountNearby, nearestBullyDistance);
782 // Annoying classmate audio disabled: no one-shot SFX played here.
783 SoundManager::UpdateGuardProximityLoop(nearestGuardDistance);
784 SoundManager::UpdateCCTVHumLoop(nearestCCTVDistance);
785 SoundManager::UpdateTerminalBuzzLoop(nearestTerminalDistance);
786 // NPC annoying-classmate audio disabled: do not start or stop any
787 // annoying-classmate audio channels from the scene.
788
789 // Apply camera shake based on guard proximity
790 const float shakeDistance = cfg.guardShakeDistance;
791 if (nearestGuardDistance < shakeDistance && playerCam)
792 {
793 auto cameraShake = playerCam->gameObject->GetComponent<CameraShake>();
794 if (!cameraShake)
795 {
796 cameraShake = playerCam->gameObject->AddComponent<CameraShake>();
797 cameraShake->Initialize();
798 }
799
800 // Intensity increases as guard gets closer
801 float intensity = (1.0f - (nearestGuardDistance / shakeDistance)) * cfg.guardShakeIntensityScale;
802 cameraShake->Shake(cfg.guardShakeDuration, intensity);
803 }
804
805 // Update alert level display
806 int guardCount = 0;
807 for (auto *g : guards)
808 {
809 if (g && (g->IsInChaseState() || g->IsInSearchState()))
810 {
811 guardCount++;
812 }
813 }
814
815 // Determine alert stage (1-4 based on escalation)
816 int alertStage = 1;
817 if (fullEscalationActive)
818 alertStage = 4;
819 else if (priorityZoneHotTimer > fullEscalationThreshold * cfg.escalationStage3Ratio)
820 alertStage = 3;
821 else if (priorityZoneHotTimer > fullEscalationThreshold * cfg.escalationStage2Ratio)
822 alertStage = 2;
823 // Play full escalation SFX immediately when escalation triggers
824 static bool escalationSfxPlayed = false;
825 if (fullEscalationActive && !escalationSfxPlayed)
826 {
827 SoundManager::PlaySFX("full_escalation");
828 escalationSfxPlayed = true;
829 }
830 if (!fullEscalationActive)
831 escalationSfxPlayed = false;
832
833 playerHUD->UpdateAlertLevel(alertStage, priorityZoneId, guardCount);
834 }
835
836 // Heatmap update
837 {
838 int tx = (int)playerTf->position.x;
839 int ty = (int)playerTf->position.y;
840
841 heatmap.AddAt(tx, ty, heatAddPerSec * Engine::Time::deltaTime);
842
843 float f = std::pow(heatDecayPerSec, Engine::Time::deltaTime);
844 heatmap.Decay(f);
845 }
846
847 // CCTV spikes + notify guards (with cooldown)
848 for (size_t i = 0; i < cctvSensors.size(); ++i)
849 {
850 auto *s = cctvSensors[i];
851 if (!s)
852 continue;
853
854 if (cctvCooldown[i] > 0.0f)
855 cctvCooldown[i] -= Engine::Time::deltaTime;
856
857 if (s->hasLastSeen)
858 {
859 int tx = (int)s->lastSeenWorld.x;
860 int ty = (int)s->lastSeenWorld.y;
861 int zid = GetZoneId(tx, ty);
862
863 if (zid >= 1 && zid <= 4)
864 {
865
866 // ALWAYS show in visualizer (real-time log)
867 if (banditVisualizer)
868 {
869 banditVisualizer->SetCCTVStatus(
870 zid,
871 s->lastSeenWorld.x,
872 s->lastSeenWorld.y,
873 true);
874 }
875
876 // ONLY spike score if cooldown allows
877 if (cctvCooldown[i] <= 0.0f)
878 {
879 cctvScore[zid] += cctvSpikeAmount;
880
881 priorityZoneId = zid;
882 ApplyPriorityZone_AssistNearest(zid);
883
884 for (auto *g : guards)
885 if (g)
886 g->OnCCTVSpotted(s->lastSeenWorld);
887
888 cctvCooldown[i] = cctvSpikeCooldown;
889 }
890 }
891 }
892
893 // clear every frame so it doesn't spam
894 s->hasLastSeen = false;
895 }
896
897 // Bandit tick
898 banditTimer += Engine::Time::deltaTime;
899 if (banditTimer >= banditInterval)
900 {
901 banditTimer = 0.0f;
902 BanditTick();
903 }
904
905 // Update visualizer selected zone and summary when bandit changes
906 if (banditVisualizer)
907 {
908 banditVisualizer->SetSelectedZone(prevChosenZoneId);
909 }
910
911 // ML summary every 5 sec only when debug on
912 if (debugML)
913 {
914 mlSummaryTimer += Engine::Time::deltaTime;
915 if (mlSummaryTimer >= cfg.mlSummaryInterval)
916 {
917 mlSummaryTimer = 0.0f;
918 PrintMLSummary();
919 }
920 }
921
922 // Win check
923 Vector2 p(playerTf->position.x, playerTf->position.y);
924 float dx = exitPos.x - p.x;
925 float dy = exitPos.y - p.y;
926
927 if (!printedWin && (dx * dx + dy * dy) <= winRadius * winRadius)
928 {
929 printedWin = true;
930 gameEnded = true;
931 // Stop any chase loops when player wins/escapes
932 SoundManager::StopLoop("guard_chasing");
933 SoundManager::PlaySFX("win");
934
935 // Update high score if this is a new record
936 float currentTime = playerData->GetGameTime();
937 if (currentTime < playerData->GetHighScore())
938 {
939 playerData->SetHighScore(currentTime);
940 }
941
942 // Persist high score to disk
943 if (playerData)
944 {
945 Json::Value root;
946 root["highScore"] = playerData->GetHighScore();
947 std::ofstream ofs("Assets/Config/Game/highscore.json");
948 if (ofs.is_open())
949 {
950 Json::StreamWriterBuilder writer;
951 ofs << Json::writeString(writer, root);
952 ofs.close();
953 }
954 }
955
956 // Trigger win state and show win screen
957 playerData->Win();
958 if (winScreen)
959 {
960 winScreen->Show();
961 }
962 }
963 }
